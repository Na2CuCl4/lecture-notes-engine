% XeTeX 2024 can resolve the family name to a variable font that xdvipdfmx
% cannot embed. Prefer the static files when that installation provides them.
\IfFontExistsTF{NotoSerifSC-Regular.ttf}{
  \setCJKmainfont{NotoSerifSC-Regular.ttf}[BoldFont=NotoSerifSC-Bold.ttf,ItalicFont=\lectureitalicfont,BoldItalicFont=\lectureitalicfont]
}{
  \IfFontExistsTF{Noto Serif SC}{
    \setCJKmainfont{Noto Serif SC}[ItalicFont={[\lectureitalicfont]},BoldItalicFont={[\lectureitalicfont]}]
  }{
    \IfFontExistsTF{Noto Serif CJK SC}{
      % With a system-family main font, TeX Live 2022 needs an explicit file
      % lookup for its bundled italic; a bare filename is treated as a family.
      \setCJKmainfont{Noto Serif CJK SC}[ItalicFont={[\lectureitalicfont]},BoldItalicFont={[\lectureitalicfont]}]
    }{} % ctex's fandol fontset already provides Song/Bold/Kai fallbacks.
